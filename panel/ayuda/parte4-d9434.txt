/* ---------- sesiones ---------- */
function firmar_(texto) {
  var secreto = PropertiesService.getScriptProperties().getProperty('SECRETO');
  return Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(texto, secreto));
}
function crearToken_(u) {
  var carga = Utilities.base64EncodeWebSafe(JSON.stringify({ u: u.nombre, r: u.rol, e: Date.now() + DIAS_SESION * 86400000 }));
  return carga + '.' + firmar_(carga);
}
function verificar_(token) {
  if (!token || String(token).indexOf('.') < 0) return null;
  var p = String(token).split('.');
  if (firmar_(p[0]) !== p[1]) return null;
  try {
    var d = JSON.parse(Utilities.newBlob(Utilities.base64DecodeWebSafe(p[0])).getDataAsString());
    if (!d.e || d.e < Date.now()) return null;
    var u = usuarios_().filter(function (x) { return x.nombre === d.u && x.activo !== 'No' })[0];
    return u ? { nombre: u.nombre, rol: u.rol } : null;
  } catch (e) { return null }
}
function usuarios_() {
  var h = hoja_('Usuarios'), cab = PanelCore.SCHEMA.Usuarios;
  if (!h || h.getLastRow() < 2) return [];
  return h.getRange(2, 1, h.getLastRow() - 1, cab.length).getValues().map(function (f) {
    return { nombre: String(f[0]).trim(), rol: String(f[1]).trim().toLowerCase() === 'admin' ? 'admin' : 'vendedor', pin: String(f[2]).trim(), activo: String(f[3]).trim() };
  }).filter(function (u) { return u.nombre });
}
function entrar_(req) {
  var cache = CacheService.getScriptCache();
  var fallos = parseInt(cache.get('fallos') || '0', 10);
  if (fallos >= 8) return { ok: false, error: 'Demasiados intentos. Espera 10 minutos y vuelve a probar.', code: 'bloqueo' };
  var pin = String(req.pin || '').trim();
  var u = pin.length >= 4 ? usuarios_().filter(function (x) { return x.pin === pin && x.activo !== 'No' })[0] : null;
  if (!u) { cache.put('fallos', String(fallos + 1), 600); return { ok: false, error: 'Clave incorrecta. Revísala e inténtalo de nuevo.', code: 'clave' } }
  cache.remove('fallos');
  return { ok: true, data: { token: crearToken_(u), user: { nombre: u.nombre, rol: u.rol } } };
}

/* ---------- entrada web ---------- */
function ahora_() {
  var d = new Date();
  return { today: Utilities.formatDate(d, ZONA, 'yyyy-MM-dd'), now: Utilities.formatDate(d, ZONA, 'yyyy-MM-dd HH:mm') };
}
function responder_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
function doGet() { return ContentService.createTextOutput('Panel GEA activo.') }
function doPost(e) {
  var salida;
  try {
    var req = JSON.parse(e.postData.contents);
    salida = ruta_(req);
  } catch (err) {
    salida = { ok: false, error: 'Algo falló en el servidor. Inténtalo otra vez. (' + err.message + ')', code: 'servidor' };
  }
  return responder_(salida);
}

/* ---------- asistente con IA (solo administrador) ---------- */
function iaPreguntar_(req, user) {
  if (user.rol !== 'admin') return { ok: false, error: 'El asistente es solo para el administrador.', code: 'permiso' };
  var props = PropertiesService.getScriptProperties(), clave = props.getProperty('CLAVE_IA');
  if (!clave) return { ok: false, error: 'Falta conectar la clave de IA. En Apps Script: Configuración del proyecto > Propiedades de la secuencia de comandos > agrega CLAVE_IA.', code: 'ia' };
  var pregunta = String(req.pregunta || '').trim().slice(0, 600);
  if (!pregunta) return { ok: false, error: 'Escribe tu pregunta.', code: 'datos' };
  var limite = parseInt(props.getProperty('LIMITE_IA_DIA') || '40', 10), dia = ahora_().today, llave = 'IA_' + dia, usadas = parseInt(props.getProperty(llave) || '0', 10);
  if (usadas >= limite) return { ok: false, error: 'Llegaste al límite de ' + limite + ' preguntas de hoy. Mañana se renueva.', code: 'limite' };
  var msgs = [];
  (req.historial || []).slice(-8).forEach(function (m) { if (m && (m.role === 'user' || m.role === 'assistant') && m.content) msgs.push({ role: m.role, content: String(m.content).slice(0, 1500) }) });
  while (msgs.length && msgs[0].role !== 'user') msgs.shift();
  if (msgs.length && msgs[msgs.length - 1].role === 'user' && msgs[msgs.length - 1].content === pregunta) msgs.pop();
  var sistema = 'Eres el asistente financiero y de operaciones de GEA Carbón (briquetas de cáscara de coco, ECOCOCO) en Retalhuleu, Guatemala, y hablas con Manolo, el dueño. Responde en español sencillo y directo, con la conclusión o la cifra primero y después una explicación corta. Usa quetzales (Q). '
    + 'Usa ÚNICAMENTE los datos del contexto; no inventes cifras. Si falta un dato, dilo y di qué debe registrar para poder responder. Las ganancias y el costo por kg son estimados que dependen de los gastos registrados; aclara cuando sea relevante. '
    + 'Cuando pida planes o estrategias, da pocas acciones concretas, baratas y fabricables o ejecutables localmente, en orden de prioridad, y evita complejidad que no tenga justificación económica. Máximo 180 palabras, sin tablas.\n\nDATOS ACTUALES DEL NEGOCIO:\n' + String(req.contexto || '').slice(0, 7000);
  msgs.push({ role: 'user', content: pregunta });
  var r;
  try {
    r = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', { method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      headers: { 'x-api-key': clave, 'anthropic-version': '2023-06-01' },
      payload: JSON.stringify({ model: props.getProperty('MODELO_IA') || 'claude-haiku-5-5', max_tokens: 700, system: sistema, messages: msgs }) });
  } catch (e) { return { ok: false, error: 'No se pudo contactar a la IA. Inténtalo de nuevo.', code: 'ia' } }
  var code = r.getResponseCode(), cuerpo = {};
  try { cuerpo = JSON.parse(r.getContentText()) } catch (e) {}
  if (code !== 200) {
    var det = cuerpo && cuerpo.error && cuerpo.error.message ? cuerpo.error.message : 'error ' + code;
    return { ok: false, error: (code === 401 ? 'La clave de IA no es válida. ' : code === 429 ? 'La IA está saturada o sin crédito. ' : 'La IA no respondió. ') + '(' + det + ')', code: 'ia' };
  }
  props.setProperty(llave, String(usadas + 1));
  var texto = (cuerpo.content || []).map(function (b) { return b.text || '' }).join('').trim();
  return { ok: true, data: { respuesta: texto || 'No obtuve respuesta. Inténtalo de nuevo.' } };
}

function ruta_(req) {
  if (req.action === 'ping') return { ok: true, data: 'Panel GEA' };
  if (req.action === 'login') return entrar_(req);
  var user = verificar_(req.token);
  if (!user) return { ok: false, error: 'Tu sesión venció. Entra de nuevo con tu clave.', code: 'auth' };
  if (req.action === 'ia.preguntar') return iaPreguntar_(req, user);
  var lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try {
    var db = leer_({}), t = ahora_();
    var id = PropertiesService.getScriptProperties().getProperty('HOJA');
    var url = id ? SpreadsheetApp.openById(id).getUrl() : '';
    var res = PanelCore.handle(db, req, { user: user, today: t.today, now: t.now, sheetUrl: url });
    if (res.ok && res.ops && res.ops.length) aplicar_(res.ops, db);
    delete res.ops;
    return res;
  } finally { lock.releaseLock() }
}
